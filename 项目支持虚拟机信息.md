services:
  mysql:
    image: mysql:8.0.43
    container_name: infra-mysql
    restart: unless-stopped
    environment:
      MYSQL_ROOT_PASSWORD: 123456
      MYSQL_DATABASE: edurag
      TZ: Asia/Shanghai
    ports:
      - "3306:3306"
    command:
      - --character-set-server=utf8mb4
      - --collation-server=utf8mb4_unicode_ci
      - --innodb-buffer-pool-size=1G
      - --local-infile=1
    volumes:
      - /root/docker-infra/data/mysql:/var/lib/mysql
    deploy:
      resources:
        limits:
          memory: 1536M
          cpus: '1.0'
        reservations:
          cpus: '0.5'
          memory: 768M

  redis:
    image: redis:8.2
    container_name: infra-redis
    restart: unless-stopped
    ports:
      - "6379:6379"
    volumes:
      - /root/docker-infra/data/redis:/data
    command: redis-server --appendonly yes --requirepass 123456
    deploy:
      resources:
        limits:
          memory: 256M
          cpus: '0.5'
        reservations:
          cpus: '0.25'
          memory: 128M

  etcd:
    image: quay.m.daocloud.io/coreos/etcd:v3.5.5
    container_name: stack-etcd
    environment:
      - ETCD_AUTO_COMPACTION_MODE=revision
      - ETCD_AUTO_COMPACTION_RETENTION=1000
      - ETCD_QUOTA_BACKEND_BYTES=4294967296
      - ETCD_SNAPSHOT_COUNT=50000
    volumes:
      - /root/docker-infra/data/etcd:/etcd
    ports:
      - "2379:2379"
    command:
      - etcd
      - --data-dir=/etcd
      - --listen-client-urls=http://0.0.0.0:2379
      - --advertise-client-urls=http://etcd:2379
    restart: unless-stopped
    deploy:
      resources:
        limits:
          memory: 256M
          cpus: '0.5'

  minio:
    image: quay.m.daocloud.io/minio/minio:RELEASE.2025-06-13T11-33-47Z
    container_name: stack-minio
    environment:
      - MINIO_ACCESS_KEY=minioadmin
      - MINIO_SECRET_KEY=minioadmin
    command: server /data --console-address ":9001"
    volumes:
      - /root/docker-infra/data/minio:/data
    ports:
      - "9000:9000"
      - "9001:9001"
    restart: unless-stopped
    deploy:
      resources:
        limits:
          memory: 512M
          cpus: '0.5'
        reservations:
          cpus: '0.25'
          memory: 256M

  milvus:
    image: milvusdb/milvus:v2.6.6
    container_name: stack-milvus
    depends_on:
      - etcd
      - minio
    ports:
      - "19530:19530"
      - "9091:9091"
    volumes:
      - /root/docker-infra/data/milvus:/var/lib/milvus
    environment:
      - ETCD_ENDPOINTS=etcd:2379
      - MINIO_ADDRESS=minio:9000
      - MINIO_ACCESS_KEY=minioadmin
      - MINIO_SECRET_KEY=minioadmin
      - MILVUS_LOG_LEVEL=info
    command: ["milvus", "run", "standalone"]
    restart: unless-stopped
    deploy:
      resources:
        limits:
          memory: 2048M
          cpus: '2.0'
        reservations:
          cpus: '1.0'
          memory: 1024M

  attu:
    image: zilliz/attu:v2.6
    container_name: stack-attu
    depends_on:
      - milvus
    environment:
      - MILVUS_URL=milvus:19530
    ports:
      - "3000:3000"
    restart: unless-stopped
    deploy:
      resources:
        limits:
          memory: 256M
          cpus: '0.5'
